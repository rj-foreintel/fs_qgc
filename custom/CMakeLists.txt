# ============================================================================
# ForeView custom build
#
# App name/IDs come from cmake/CustomOverrides.cmake. Logos and the
# customer-mode QML overrides are swapped in through ForeViewPlugin's
# resource interceptor (see custom.qrc), and Android launcher icons are
# overlaid on the stock android/ package template.
# ============================================================================

message(STATUS "ForeView: adding custom build layer")

# ----------------------------------------------------------------------------
# Android package template: stock android/ + our res/ overlay
# ----------------------------------------------------------------------------
if(ANDROID)
    set(_default_android_dir "${CMAKE_SOURCE_DIR}/android")
    set(_custom_android_dir  "${CMAKE_CURRENT_SOURCE_DIR}/android")
    set(_final_android_dir   "${CMAKE_BINARY_DIR}/custom/android")

    # copy_directory always overwrites. file(COPY) would skip our icons whenever
    # they share a timestamp with the stock ones, which is the norm after a git checkout.
    file(REMOVE_RECURSE "${_final_android_dir}")
    execute_process(COMMAND ${CMAKE_COMMAND} -E copy_directory "${_default_android_dir}" "${_final_android_dir}" COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND ${CMAKE_COMMAND} -E copy_directory "${_custom_android_dir}" "${_final_android_dir}" COMMAND_ERROR_IS_FATAL ANY)

    # The stock build.gradle never sets applicationId, so Gradle would fall back
    # to the "org.mavlink.qgroundcontrol" namespace and ForeView would collide
    # with stock QGC on the device. Pin it to our package name.
    set(_gradle_file "${_final_android_dir}/build.gradle")
    file(READ "${_gradle_file}" _gradle)
    string(FIND "${_gradle}" "    defaultConfig {" _pos)
    if(_pos EQUAL -1)
        message(FATAL_ERROR "ForeView: could not find defaultConfig in ${_gradle_file}; update custom/CMakeLists.txt")
    endif()
    string(REPLACE "    defaultConfig {"
                   "    defaultConfig {\n        applicationId = \"${QGC_ANDROID_PACKAGE_NAME}\""
                   _gradle "${_gradle}")
    file(WRITE "${_gradle_file}" "${_gradle}")

    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${_default_android_dir}/build.gradle"
        "${_default_android_dir}/AndroidManifest.xml"
    )

    set(QGC_ANDROID_PACKAGE_SOURCE_DIR "${_final_android_dir}" CACHE PATH "Path to a custom Android package template" FORCE)
    message(STATUS "ForeView: Android package template at ${QGC_ANDROID_PACKAGE_SOURCE_DIR} (applicationId ${QGC_ANDROID_PACKAGE_NAME})")
endif()

# ----------------------------------------------------------------------------
# Resources (logos looked up under :/Custom/... by the interceptor)
# ----------------------------------------------------------------------------
set(QGC_RESOURCES ${QGC_RESOURCES} ${CMAKE_CURRENT_SOURCE_DIR}/custom.qrc CACHE STRING "Paths to .qrc Resources" FORCE)

# ----------------------------------------------------------------------------
# Debug Mode password
#
# The full QGroundControl UI ("Debug Mode") is unlocked with a company
# password, passed in at configure time as FOREVIEW_DEBUG_PASSWORD (CI reads
# it from the repository secret of the same name). Only a random salt and a
# stretched SHA-256 hash are compiled in; ForeViewMode.cc repeats the same
# hashing to check what the user types.
# ----------------------------------------------------------------------------
set(_foreview_password "$ENV{FOREVIEW_DEBUG_PASSWORD}")
if(_foreview_password STREQUAL "")
    if("$ENV{FOREVIEW_REQUIRE_DEBUG_PASSWORD}" STREQUAL "true")
        message(FATAL_ERROR "ForeView: FOREVIEW_DEBUG_PASSWORD is not set. Add it as a repository secret before building a release.")
    endif()
    set(_foreview_password "foreview-dev")
    message(WARNING "ForeView: FOREVIEW_DEBUG_PASSWORD is not set; Debug Mode uses the development password \"foreview-dev\". Do not give this build to customers.")
endif()

set(_foreview_iterations 20000)
string(RANDOM LENGTH 32 ALPHABET "0123456789abcdef" _foreview_salt)
string(SHA256 _foreview_hash "${_foreview_salt}${_foreview_password}")
foreach(_i RANGE 2 ${_foreview_iterations})
    string(SHA256 _foreview_hash "${_foreview_hash}${_foreview_salt}")
endforeach()
unset(_foreview_password)

# A generated header (included only by ForeViewMode.cc) rather than compile definitions,
# so a new salt does not change the command line of every other source file.
set(_foreview_generated_dir "${CMAKE_BINARY_DIR}/foreview")
file(WRITE "${_foreview_generated_dir}/ForeViewDebugPassword.h"
    "// Generated by custom/CMakeLists.txt. Do not edit.\n"
    "#pragma once\n"
    "#define FOREVIEW_DEBUG_SALT \"${_foreview_salt}\"\n"
    "#define FOREVIEW_DEBUG_HASH \"${_foreview_hash}\"\n"
    "#define FOREVIEW_DEBUG_ITERATIONS ${_foreview_iterations}\n"
)

# ----------------------------------------------------------------------------
# Core plugin
# ----------------------------------------------------------------------------
set(CUSTOM_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewCamera.cc
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewCamera.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewMode.cc
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewMode.h
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewPlugin.cc
    ${CMAKE_CURRENT_SOURCE_DIR}/src/ForeViewPlugin.h
    CACHE INTERNAL "" FORCE
)

set(CUSTOM_INCLUDE_DIRECTORIES
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${_foreview_generated_dir}
    CACHE INTERNAL "" FORCE
)

set(CUSTOM_DEFINITIONS
    QGC_CUSTOM_BUILD
    CUSTOMHEADER="ForeViewPlugin.h"
    CUSTOMCLASS=ForeViewPlugin
    CACHE INTERNAL "" FORCE
)

set(CUSTOM_QT_COMPONENTS
    Core
    Network
    Qml
    CACHE INTERNAL "" FORCE
)
